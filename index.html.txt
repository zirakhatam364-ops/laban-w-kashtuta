            updateUIForRole(); // Check login state on load
        }

        // --- SYSTEM FUNCTIONS ---
        function showToast(message, type = 'info') {
            const container = document.getElementById('toast-container');
            const toast = document.createElement('div');
            toast.className = `toast ${type}`;
            let icon = 'info-circle';
            if(type === 'success') icon = 'check-circle';
            if(type === 'error') icon = 'exclamation-circle';
            toast.innerHTML = `<i class="fas fa-${icon}"></i> ${message}`;
            container.appendChild(toast);
            setTimeout(() => { toast.remove(); }, 3000);
        }

        function toggleDarkMode() {
            isDarkMode = !isDarkMode;
            document.body.classList.toggle('dark-mode');
            localStorage.setItem('darkMode', isDarkMode);
        }

        function setLanguage(lang) {
            currentLang = lang;
            document.body.classList.remove('rtl');
            if (lang === 'ar') document.body.classList.add('rtl');
            
            document.querySelectorAll('[data-translate]').forEach(el => {
                const key = el.getAttribute('data-translate');
                if (translations[lang][key]) el.innerText = translations[lang][key];
            });

            document.getElementById('searchInput').placeholder = translations[lang].search_placeholder;
            renderMenu();
            renderCategoryList();
        }

        // --- PERMISSIONS & LOGIN ---
        function toggleLogin() {
            if (isManagerLoggedIn) {
                isManagerLoggedIn = false;
                showToast("Logged Out", 'info');
            } else {
                let input = prompt("Enter Manager Password:");
                if (input === MANAGER_PASSWORD) {
                    isManagerLoggedIn = true;
                    showToast(translations[currentLang].login_success, 'success');
                } else if (input !== null) {
                    showToast(translations[currentLang].wrong_pass, 'error');
                    return;
                }
            }
            updateUIForRole();
            renderMenu();
        }

        function updateUIForRole() {
            const adminBtns = document.querySelectorAll('.admin-only');
            const authBtn = document.getElementById('authBtn');
            const downloadQrBtn = document.getElementById('downloadQrBtn');
            const settingsFloat = document.getElementById('settingsBtnFloating');

            if (isManagerLoggedIn) {
                adminBtns.forEach(btn => btn.style.display = 'flex');
                authBtn.classList.add('active');
                authBtn.innerHTML = `<i class="fas fa-sign-out-alt"></i> <span>${translations[currentLang].logout}</span>`;
                if(downloadQrBtn) downloadQrBtn.style.display = 'inline-flex';
                if(settingsFloat) settingsFloat.style.display = 'flex';
            } else {
                adminBtns.forEach(btn => btn.style.display = 'none');
                authBtn.classList.remove('active');
                authBtn.innerHTML = `<i class="fas fa-user-lock"></i> <span>${translations[currentLang].login}</span>`;
                if(downloadQrBtn) downloadQrBtn.style.display = 'none';
                if(settingsFloat) settingsFloat.style.display = 'none';
            }
        }

        // --- SEARCH ---
        function searchMenu() {
            const input = document.getElementById('searchInput').value.toLowerCase();
            const cards = document.querySelectorAll('.card');
            cards.forEach(card => {
                const title = card.querySelector('.card-title').innerText.toLowerCase();
                const desc = card.querySelector('.card-desc')?.innerText.toLowerCase() || "";
                if (title.includes(input) || desc.includes(input)) {
                    card.style.display = 'flex';
                } else {
                    card.style.display = 'none';
                }
            });
        }

        // --- DATA MANAGEMENT ---
        function exportToExcel() {
            let csvContent = "data:text/csv;charset=utf-8,\uFEFF";
            csvContent += "Name,Category,Price (IQD),Description,Image URL,Available\n";
            menuItems.forEach(item => {
                const name = `"${item.name.replace(/"/g, '""')}"`;
                const category = `"${item.category.replace(/"/g, '""')}"`;
                const price = item.price;
                const desc = `"${item.desc.replace(/"/g, '""')}"`;
                const img = `"${item.img}"`;
                const available = item.available ? "Yes" : "No";
                csvContent += `${name},${category},${price},${desc},${img},${available}\n`;
            });
            const encodedUri = encodeURI(csvContent);
            const link = document.createElement("a");
            link.setAttribute("href", encodedUri);
            link.setAttribute("download", "menu_export.csv");
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
            showToast("Excel File Exported!", 'success');
        }

        function closeSettingsModal() { document.getElementById('settingsModalOverlay').classList.remove('active'); }
        function closeSettingsModalOutside(e) { if(e.target.id === 'settingsModalOverlay') closeSettingsModal(); }

        function exportData() {
            const data = { categories: categories, items: menuItems };
            const blob = new Blob([JSON.stringify(data)], { type: "application/json" });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = 'menu_backup.json';
            a.click();
            showToast("Backup Downloaded!", 'success');
        }

        function importData(input) {
            if (input.files && input.files[0]) {
                const reader = new FileReader();
                reader.onload = function (e) {
                    try {
                        const data = JSON.parse(e.target.result);
                        if(data.categories && data.items) {
                            categories = data.categories;
                            menuItems = data.items;
                            saveCategories();
                            saveData();
                            renderMenu();
                            renderCategoryList();
                            showToast("Data Imported Successfully!", 'success');
                        } else {
                            showToast("Invalid File Format", 'error');
                        }
                    } catch(err) {
                        showToast("Error Reading File", 'error');
                    }
                };
                reader.readAsText(input.files[0]);
            }
        }

        // --- CATEGORY FUNCTIONS ---
        function openCategoryModal() { if(isManagerLoggedIn){ renderCategoryList(); document.getElementById('catModalOverlay').classList.add('active'); } }
        function closeCategoryModal() { document.getElementById('catModalOverlay').classList.remove('active'); }
        function closeCategoryModalOutside(e) { if(e.target.id === 'catModalOverlay') closeCategoryModal(); }

        function renderCategoryList() {
            const list = document.getElementById('categoryList');
            list.innerHTML = '';
            categories.forEach((cat, index) => {
                list.innerHTML += `<div class="cat-item" style="display:flex; justify-content:space-between; align-items:center; padding:10px; background:var(--bg-color); margin-bottom:5px; border-radius:5px;"><span>${cat}</span><button class="admin-btn delete-btn" onclick="deleteCategory(${index})"><i class="fas fa-trash"></i></button></div>`;
            });
        }

        function addCategory() {
            if(!isManagerLoggedIn) return;
            const input = document.getElementById('newCatName');
            const name = input.value.trim();
            if (name) {
                categories.push(name);
                saveCategories();
                input.value = '';
                renderCategoryList();
                populateCategoryDropdown();
                renderMenu();
                showToast("Category Added!", 'success');
            }
        }

        function deleteCategory(index) {
            if(!isManagerLoggedIn) return;
            if(confirm(translations[currentLang].confirm_delete)) {
                categories.splice(index, 1);
                saveCategories();
                renderCategoryList();
                populateCategoryDropdown();
                renderMenu();
            }
        }

        function populateCategoryDropdown() {
            const select = document.getElementById('itemCategory');
            select.innerHTML = '';
            categories.forEach(cat => { select.innerHTML += `<option value="${cat}">${cat}</option>`; });
        }

        function scrollToCategory(categoryName) {
            const sectionId = 'section-' + categoryName.replace(/\s+/g, '-');
            const section = document.getElementById(sectionId);
            if (section) {
                section.scrollIntoView({ behavior: 'smooth', block: 'start' });
                document.querySelectorAll('.cat-pill').forEach(btn => {
                    btn.classList.remove('active');
                    if(btn.innerText === categoryName) btn.classList.add('active');
                });
            }
        }

        // --- ITEM FUNCTIONS ---
        function openModal() {
            if(!isManagerLoggedIn) return;
            populateCategoryDropdown();
            document.getElementById('modalOverlay').classList.add('active');
            document.getElementById('modalTitle').innerText = translations[currentLang].add_new;
            document.getElementById('itemForm').reset();
            document.getElementById('imagePreview').style.display = 'none';

        }

        function closeModal() { document.getElementById('modalOverlay').classList.remove('active'); }

        function saveItem(e) {
            if(!isManagerLoggedIn) return;
            e.preventDefault();
            const name = document.getElementById('itemName').value;
            const category = document.getElementById('itemCategory').value;
            let img = document.getElementById('itemImageHidden').value;
            const price = parseFloat(document.getElementById('itemPrice').value);
            const desc = document.getElementById('itemDesc').value;

            if (editingId && !img) {
                const oldItem = menuItems.find(i => i.id === editingId);
                if(oldItem) img = oldItem.img;
            }

            if (editingId) {
                const index = menuItems.findIndex(i => i.id === editingId);
                const currentStatus = menuItems[index].available;
                menuItems[index] = { id: editingId, name, category, img, price, desc, available: currentStatus };
                showToast("Item Updated!", 'success');
            } else {
                if(!img) img = 'https://via.placeholder.com/300x200?text=No+Image';
                menuItems.push({ id: Date.now(), name, category, img, price, desc, available: true });
                showToast("Item Added!", 'success');
            }

            saveData();
            renderMenu();
            closeModal();
        }

        function editItem(id) {
            if(!isManagerLoggedIn) return;
            populateCategoryDropdown();
            const item = menuItems.find(i => i.id === id);
            editingId = id;
            
            document.getElementById('modalTitle').innerText = translations[currentLang].edit;
            document.getElementById('itemName').value = item.name;
            document.getElementById('itemCategory').value = item.category;
            document.getElementById('itemPrice').value = item.price;
            document.getElementById('itemDesc').value = item.desc;
            document.getElementById('itemImageHidden').value = item.img;
            
            if(item.img) {
                document.getElementById('imagePreview').src = item.img;
                document.getElementById('imagePreview').style.display = 'block';
            }
            document.getElementById('modalOverlay').classList.add('active');
        }

        function deleteItem(id) {
            if(!isManagerLoggedIn) return;
            if(confirm(translations[currentLang].confirm_delete)) {
                menuItems = menuItems.filter(i => i.id !== id);
                saveData();
                renderMenu();
                showToast("Item Deleted", 'info');
            }
        }

        function toggleAvailability(id) {
            if(!isManagerLoggedIn) return;
            const item = menuItems.find(i => i.id === id);
            if(item) {
                item.available = !item.available;
                saveData();
                renderMenu();
                showToast(item.available ? "Item Available" : "Item Sold Out", 'info');
            }
        }

        // --- RENDERING ---
        function renderMenu() {
            const navContainer = document.getElementById('categoryNav');
            const menuContainer = document.getElementById('menuContainer');
            
            navContainer.innerHTML = '';
            categories.forEach(cat => {
                const pill = document.createElement('button');
                pill.className = 'cat-pill';
                pill.innerText = cat;
                pill.onclick = () => scrollToCategory(cat);
                navContainer.appendChild(pill);
            });

            menuContainer.innerHTML = '';
            
            categories.forEach(cat => {
                const itemsInCat = menuItems.filter(item => item.category === cat);
                if (itemsInCat.length === 0) return;

                const section = document.createElement('div');
                section.className = 'category-section';
                section.id = 'section-' + cat.replace(/\s+/g, '-');
                
                let gridHTML = '';
                itemsInCat.forEach((item, index) => {
                    const delay = index * 0.1;
                    const isAvailable = item.available !== false;

                    let adminControlsHTML = '';
                    if (isManagerLoggedIn) {
                        adminControlsHTML = `
                            <div class="admin-controls">
                                <button class="admin-btn stock-btn ${!isAvailable ? 'inactive' : ''}" onclick="toggleAvailability(${item.id})" title="Toggle Stock">
                                    <i class="fas fa-${isAvailable ? 'check' : 'times'}"></i>
                                </button>
                                <button class="admin-btn edit-btn" onclick="editItem(${item.id})"><i class="fas fa-pen"></i></button>
                                <button class="admin-btn delete-btn" onclick="deleteItem(${item.id})"><i class="fas fa-trash"></i></button>
                            </div>
                        `;
                    }
                    
                    gridHTML += `
                        <div class="card ${isManagerLoggedIn ? 'show-admin' : ''} ${!isAvailable ? 'sold-out' : ''}" style="animation-delay: ${delay}s;">
                            <div class="card-img-container">
                                <img src="${item.img}" alt="${item.name}" class="card-img" onerror="this.src='https://via.placeholder.com/300x200?text=No+Image'">
                            </div>
                            
                            ${!isAvailable ? `<div class="sold-out-badge">${translations[currentLang].sold_out}</div>` : ''}

                            ${adminControlsHTML}

                            <div class="card-content">
                                <h3 class="card-title">${item.name}</h3>
                                <p class="card-desc">${item.desc}</p>
                                <p class="card-price">${item.price.toLocaleString()} د.ع</p>
                            </div>
                        </div>
                    `;
                });

                section.innerHTML = `<h3 class="category-title" style="font-size: 1.5rem; color: var(--primary); font-family: 'Playfair Display', serif; margin-bottom: 15px; border-left: 4px solid var(--secondary); padding-left: 10px;">${cat}</h3><div class="menu-grid">${gridHTML}</div>`;
                menuContainer.appendChild(section);
            });
        }

        // --- IMAGE HANDLING ---
        function previewImage(input) {
            if (input.files && input.files[0]) {
                const reader = new FileReader();
                reader.onload = function (e) {
                    const img = new Image();
                    img.src = e.target.result;
                    img.onload = function() {
                        const canvas = document.createElement('canvas');
                        const ctx = canvas.getContext('2d');
                        const maxWidth = 600;
                        let width = img.width;
                        let height = img.height;
                        
                        if (width > maxWidth) { height = height * (maxWidth / width); width = maxWidth; }
                        
                        canvas.width = width;
                        canvas.height = height;
                        ctx.drawImage(img, 0, 0, width, height);
                        
                        const finalSrc = canvas.toDataURL('image/jpeg', 0.9);
                        document.getElementById('itemImageHidden').value = finalSrc;
                        document.getElementById('imagePreview').src = finalSrc;
                        document.getElementById('imagePreview').style.display = 'block';
                    };
                };
                reader.readAsDataURL(input.files[0]);
            }
        }

        function generateQR() {

            const qrApiUrl = `https://api.qrserver.com/v1/create-qr-code/?size=200x200&data=${encodeURIComponent(currentUrl)}`;
            
            const qrImg = document.getElementById('qrCodeImage');
            qrImg.src = qrApiUrl;
            
            const downloadBtn = document.getElementById('downloadQrBtn');
            if (isManagerLoggedIn) { downloadBtn.style.display = 'inline-flex'; } 
            else { downloadBtn.style.display = 'none'; }

            document.getElementById('qrModalOverlay').classList.add('active');
        }
        
        function downloadQR() {
            const imgSrc = document.getElementById('qrCodeImage').src;
            fetch(imgSrc)
                .then(response => response.blob())
                .then(blob => {
                    const url = window.URL.createObjectURL(blob);
                    const a = document.createElement('a');
                    a.href = url;
                    a.download = 'menu-qrcode.png';
                    document.body.appendChild(a);
                    a.click();
                    document.body.removeChild(a);
                });
        }
